def read(path):
    with gzip.open(path, 'rt') if path.suffix == '.gz' else path.open() as f:
        return json.load(f)


def reach(start, adj):
    seen, pending = {start}, [start]
    while pending:
        for dst in adj[pending.pop()]:
            if dst not in seen:
                seen.add(dst)
                pending.append(dst)
    return seen


def quote(text):
    return json.dumps(str(text))


def instruction(site):
    rec = site.get('decoder_record') or ''
    parts = rec.split()
    op = site.get('opcode') or 'unavailable'
    args = parts[3:3 + int(parts[2])] if len(parts) > 2 else []
    if op == 'PUSH64r' and args:
        return 'push ' + args[0].removeprefix('r:').lower()
    if op in ('SUB64ri8', 'SUB64ri32') and len(args) == 3:
        return 'sub rsp, ' + hex(int(args[-1].removeprefix('i:')))
    if site['va'] == STOP:
        return 'movaps [rsp+0x240], xmm6'
    if site['va'] in ('0x00007ff77443d992', PC):
        return 'cmp byte ptr [rax+0x38], 0x76'
    if site['va'] == '0x00007ff77443dd66':
        return 'mov [rsi+0x18], rax'
    if site['va'] == '0x00007ff77443dd6a':
        return 'mov rax, [rsi+0x28]'
    if op == 'MOV64rr' and len(args) == 2:
        return 'mov ' + ', '.join(x.removeprefix('r:').lower() for x in args)
    def memory(values):
        base, scale, index, displacement = values[:4]
        terms = []
        if base != 'r:NONE':
            terms.append(base.removeprefix('r:').lower())
        if index != 'r:NONE':
            terms.append(index.removeprefix('r:').lower() + '*' + scale.removeprefix('i:'))
        disp = int(displacement.removeprefix('i:'))
        address = '+'.join(terms)
        if disp:
            address += ('+' if address and disp > 0 else '') + hex(disp)
        return '[' + (address or '0') + ']'
    if op == 'MOV64rm' and len(args) == 6:
        return 'mov ' + args[0].removeprefix('r:').lower() + ', ' + memory(args[1:])
    if op == 'MOV64mr' and len(args) == 6:
        return 'mov ' + memory(args[:5]) + ', ' + args[-1].removeprefix('r:').lower()
    if op == 'XOR64rr' and len(args) == 3 and args[0] == args[1]:
        return 'xor ' + args[0].removeprefix('r:').lower() + ', ' + args[2].removeprefix('r:').lower()
    if op.startswith('JCC'):
        return op + '  conditional branch'
    return op

graph_data = {}
for frame, report in reports.items():
    analysis = report['analysis']
    sites = {s['va']: s for s in report['preparation']['sites']}
    local = [e for e in analysis['edges'] if e['local']]
    nodes = set(analysis['visited']) | set(analysis['decoded'])
    succ, pred, calls = (collections.defaultdict(list) for _ in range(3))
    for e in local:
        nodes.update((e['src'], e['dst']))
        succ[e['src']].append(e)
        pred[e['dst']].append(e)
    for e in analysis['edges']:
        if not e['local']:
            calls[e['src']].append(e)
    leaders = {ENTRIES[frame]} | (set(analysis['visited']) - set(analysis['decoded']))
    if frame == 0:
        leaders.add(PC)
    for va in nodes:
        if len(pred[va]) != 1:
            leaders.add(va)
    for e in local:
        if e['kind'] != 'next' or len(succ[e['src']]) != 1 or calls[e['src']]:
            leaders.add(e['dst'])
    blocks, owner = [], {}
    ordered = sorted(leaders & nodes) + sorted(nodes - leaders)
    for start in ordered:
        if start in owner:
            continue
        block, cur = [], start
        while cur not in owner:
            owner[cur] = len(blocks)
            block.append(cur)
            edges = succ[cur]
            if len(edges) != 1 or edges[0]['kind'] != 'next' or calls[cur]:
                break
            nxt = edges[0]['dst']
            if nxt in leaders or nxt in owner:
                break
            cur = nxt
        blocks.append(block)
    assert set(owner) == nodes
    internal = [e for e in local if owner[e['src']] == owner[e['dst']]]
    between = [e for e in local if owner[e['src']] != owner[e['dst']]]
    assert len(internal) + len(between) == len(local)
    assert all(e['kind'] == 'next' for e in internal)
    graph_data[frame] = dict(sites=sites, nodes=nodes, local=local, succ=succ,
                             blocks=blocks, owner=owner, calls=calls, internal=internal)

def view(frame, included=None, individual=False, prefix=None):
    g = graph_data[frame]
    inc = set(g['nodes'] if included is None else included)
    prefix = prefix or f'f{frame}'
    groups = [[n] for n in sorted(inc)] if individual else [
        [n for n in block if n in inc] for block in g['blocks'] if any(n in inc for n in block)]
    owners = {n: i for i, block in enumerate(groups) for n in block}
    lines = []
    for i, block in enumerate(groups):
        shown = block if len(block) <= 6 else block[:2] + [None] + block[-3:]
        label = []
        for n in shown:
            if n is None:
                label.append(f'... {len(block) - 5} more instructions ...')
            else:
                label.append(n + '  ' + instruction(g['sites'].get(n, {'va': n})))
        if len(block) > 1:
            label.insert(0, f'{len(block)} retained instruction starts')
        fill, color, width = '#f4f8f8', '#537c7b', '1'
        if ENTRIES[frame] in block:
            label.insert(0, f'FRAME {frame} ENTRY')
            fill, color, width = '#e2f2ef', '#12665f', '2'
        if STOP in block:
            label += ['GUARD NOT ADMITTED', 'normal / alignment fault unresolved', 'candidate coverage incomplete']
            fill, color, width = '#fff0d9', '#b96c15', '2'
        if PC in block:
            label.insert(0, 'CAPTURED PC')
            fill, color, width = '#e4eefc', '#315f9b', '2'
        obligations = [o for o in reports[frame]['analysis']['obligations'] if o['va'] in block]
        if obligations and STOP not in block:
            label += ['OPEN: ' + ', '.join(sorted({o['reason'] for o in obligations}))]
            fill, color = '#fff0d9', '#b96c15'
        details = '\n'.join(n + '  ' + instruction(g['sites'].get(n, {'va': n})) + '\n' +
                            (g['sites'].get(n, {}).get('bytes_hex') or '') + '\n' +
                            (g['sites'].get(n, {}).get('decoder_record') or '') for n in block)
        lines.append(f'{prefix}_{i} [id="{prefix}_{i}", label={quote(chr(10).join(label))}, tooltip={quote(details)}, fillcolor="{fill}", color="{color}", penwidth={width}];')
    grouped = collections.Counter()
    hidden = collections.defaultdict(list)
    for e in g['local']:
        if e['src'] not in inc:
            continue
        if e['dst'] not in inc:
            hidden[owners[e['src']]].append(e)
        elif owners[e['src']] != owners[e['dst']]:
            grouped[(owners[e['src']], owners[e['dst']], e['kind'])] += 1
    for (a, b, kind), count in sorted(grouped.items()):
        label = kind if count == 1 else f'{kind} x{count}'
        style = ', color="#315f9b"' if kind == 'summary' else ''
        lines.append(f'{prefix}_{a} -> {prefix}_{b} [label={quote(label)}{style}];')
    for a, edges in hidden.items():
        after_pc = PC in groups[a]
        label = 'Recovered graph continues after PC' if after_pc else 'Other retained branch\n' + '\n'.join(e['kind'] + ' -> ' + e['dst'] for e in edges)
        lines.append(f'{prefix}_other{a} [shape=note, fillcolor="#fafafa", color="#9a9a9a", label={quote(label)}];')
        lines.append(f'{prefix}_{a} -> {prefix}_other{a} [color="#999999", label={quote("next" if after_pc else ", ".join(sorted({e["kind"] for e in edges})))}];')
    for src, edges in g['calls'].items():
        if src not in inc:
            continue
        for j, e in enumerate(edges):
            ext = f'{prefix}_call{owners[src]}_{j}'
            lines.append(f'{ext} [shape=ellipse, fillcolor="#f1f4fb", color="#6a7f9f", label={quote("call target " + e["dst"])}];')
            lines.append(f'{prefix}_{owners[src]} -> {ext} [style=dashed, color="#6a7f9f", label="call (display only)"];')
    return lines, owners, groups


def render(name, body):
    dot = HEADER + '\n'.join(body) + '\n}\n'
    (OUT / f'{name}.dot').write_text(dot)
    subprocess.run(['dot', '-Tsvg', str(OUT / f'{name}.dot'), '-o', str(OUT / f'{name}.svg')], check=True, timeout=45)
    ET.parse(OUT / f'{name}.svg')
    svg = (OUT / f'{name}.svg').read_text()
    return svg[svg.index('<svg'):]
