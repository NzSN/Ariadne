//! Task-private supplement against the already-qualified release library.
use ariadne::bap::core_protocol::{COMPOSED_PROFILE, LEGACY_COMPOSED_PROFILE};
use ariadne::bap::core_session::{CoreConfig, CoreSession};
use serde_json::{Value, json};
use std::error::Error;
use std::path::PathBuf;

fn main() -> Result<(), Box<dyn Error>> {
    let args: Vec<_> = std::env::args_os().skip(1).collect();
    if args.len() != 2 {
        return Err("usage: legacy-capability-probe OLD_HELPER_DIR RESULT_JSON".into());
    }
    let config = CoreConfig::from_directory(PathBuf::from(&args[0]));
    let manifest_bytes = std::fs::read(&config.manifest)?;
    let manifest: Value = serde_json::from_slice(&manifest_bytes)?;
    let helper_sha256 = ariadne::reports::sha256(&std::fs::read(&config.helper)?);
    assert_eq!(
        helper_sha256,
        "cc9ff8718659af45e3d86ba0949915615f8bdb64822638597522af63dba0c0f8"
    );
    assert_eq!(manifest["helper_sha256"], helper_sha256);
    assert_eq!(
        ariadne::reports::sha256(&manifest_bytes),
        "70d7a8ee8a6c3d30b214349208524676b1038dd9544c5f225def51a2b68c2798"
    );
    let profiles = manifest["handshake"]["profiles"]
        .as_array()
        .ok_or("profiles absent")?;
    assert!(profiles.contains(&json!(LEGACY_COMPOSED_PROFILE)));
    assert!(!profiles.contains(&json!(COMPOSED_PROFILE)));
    // Only the task-owned copy has execute permission removed. The genuine old
    // binary's bytes and manifest remain unchanged. A launch attempt could not
    // produce the expected manifest-capability rejection.
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        assert_eq!(
            std::fs::metadata(&config.helper)?.permissions().mode() & 0o111,
            0
        );
    }
    let error = match CoreSession::start_profile(
        &config,
        "loop-legacy-capability",
        "capability-snapshot",
        COMPOSED_PROFILE,
        &json!({}),
    ) {
        Err(error) => error.to_string(),
        Ok(_) => return Err("old-capability helper accepted the new combined profile".into()),
    };
    assert_eq!(
        error,
        "native helper lacks the requested input profile; rebuild the helper"
    );
    let result = json!({
        "schema":"ariadne.legacy-combined-capability-control/v1",
        "passed":true,"requested_profile":COMPOSED_PROFILE,
        "retained_profiles":profiles,"error":error,
        "rejection_stage":"manifest capability check before helper process launch",
        "helper_executable":false,"retained_helper_sha256":helper_sha256,
        "retained_manifest_sha256":ariadne::reports::sha256(&manifest_bytes),
        "scope":"Current qualified Rust adapter rejects a genuine manifest-bound old helper lacking the new profile; no historical helper runtime is exercised."
    });
    std::fs::write(&args[1], serde_json::to_vec_pretty(&result)?)?;
    println!("{result}");
    Ok(())
}
