//! Task-private capability control against the original loop-generation helper.
use ariadne::bap::core_protocol::{
    COMPOSED_PROFILE, GUARDED_PROFILE, LEGACY_GUARDED_PROFILE, LEGACY_V2_COMPOSED_PROFILE,
};
use ariadne::bap::core_session::{CoreConfig, CoreSession};
use serde_json::{Value, json};
use std::error::Error;
use std::path::PathBuf;

const OLD_HELPER: &str = "7f8030820b486dc5a7ec055433d5a7540496c3aaa85fb04c3d91cbdc9e6ec1dd";
const OLD_MANIFEST: &str = "204ffe8a8f7585a8f873886b48e64081b6f69a707604e47d758b664c6c872712";

fn main() -> Result<(), Box<dyn Error>> {
    let args: Vec<_> = std::env::args_os().skip(1).collect();
    if args.len() != 2 {
        return Err("usage: legacy-capability-probe OLD_HELPER_DIR RESULT_JSON".into());
    }
    assert_eq!(GUARDED_PROFILE, "captured-guarded-input/v2");
    assert_eq!(COMPOSED_PROFILE, "captured-guarded-targets/v3");
    let config = CoreConfig::from_directory(PathBuf::from(&args[0]));
    let manifest_bytes = std::fs::read(&config.manifest)?;
    let manifest: Value = serde_json::from_slice(&manifest_bytes)?;
    let helper_sha256 = ariadne::reports::sha256(&std::fs::read(&config.helper)?);
    assert_eq!(helper_sha256, OLD_HELPER);
    assert_eq!(manifest["helper_sha256"], helper_sha256);
    assert_eq!(ariadne::reports::sha256(&manifest_bytes), OLD_MANIFEST);
    let profiles = manifest["handshake"]["profiles"]
        .as_array()
        .ok_or("profiles absent")?;
    assert!(profiles.contains(&json!(LEGACY_GUARDED_PROFILE)));
    assert!(profiles.contains(&json!(LEGACY_V2_COMPOSED_PROFILE)));
    assert!(!profiles.contains(&json!(GUARDED_PROFILE)));
    assert!(!profiles.contains(&json!(COMPOSED_PROFILE)));
    // The copied helper retains its original bytes but is nonexecutable. An
    // attempted launch would yield a permission error, never the required
    // capability error. The original retained helper is not modified.
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        assert_eq!(
            std::fs::metadata(&config.helper)?.permissions().mode() & 0o111,
            0
        );
    }
    let mut controls = Vec::new();
    for profile in [GUARDED_PROFILE, COMPOSED_PROFILE] {
        let error = match CoreSession::start_profile(
            &config,
            "caller-legacy-capability",
            "capability-snapshot",
            profile,
            &json!({}),
        ) {
            Err(error) => error.to_string(),
            Ok(_) => return Err(format!("old helper accepted new profile {profile}").into()),
        };
        assert_eq!(
            error,
            "native helper lacks the requested input profile; rebuild the helper"
        );
        controls.push(
            json!({"requested_profile":profile,"passed":true,"error":error,
            "rejection_stage":"manifest capability check before helper process launch"}),
        );
    }
    let result = json!({
        "schema":"ariadne.caller-legacy-capability-control/v1","passed":true,
        "controls":controls,"retained_profiles":profiles,"helper_executable":false,
        "retained_helper_sha256":helper_sha256,"retained_manifest_sha256":OLD_MANIFEST,
        "current_guard_model_sha256":ariadne::bap::guard_admission::model_identity(),
        "current_guard_engine_sha256":ariadne::bap::guard_admission::checker_identity(),
        "current_composition_model_sha256":ariadne::bap::composition::model_identity(),
        "current_composition_engine_sha256":ariadne::bap::composition::checker_identity(),
        "scope":"Current public CoreSession API rejects both new modes using a genuine manifest-bound loop-generation helper before launch; no historical helper runtime or instruction semantics is exercised."
    });
    std::fs::write(&args[1], serde_json::to_vec_pretty(&result)?)?;
    println!("{result}");
    Ok(())
}
