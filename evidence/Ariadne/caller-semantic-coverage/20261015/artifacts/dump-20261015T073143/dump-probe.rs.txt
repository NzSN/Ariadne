fn va(a: u64) -> String {
    format!("0x{a:016x}")
}
fn address(s: &str) -> Result<u64, Box<dyn Error>> {
    Ok(u64::from_str_radix(s.trim_start_matches("0x"), 16)?)
}
fn write(out: &Path, name: &str, value: &Value) -> Result<(), Box<dyn Error>> {
    save(&out.join(name), &serde_json::to_vec_pretty(value)?)?;
    Ok(())
}
fn save(path: &Path, bytes: &[u8]) -> Result<PathBuf, Box<dyn Error>> {
    let destination = if bytes.len() < 1024 * 1024 {
        std::fs::write(path, bytes)?;
        path.to_path_buf()
    } else {
        let mut name = path.as_os_str().to_os_string();
        name.push(".gz");
        let destination = PathBuf::from(name);
        let mut child = Command::new("gzip")
            .args(["-9", "-c"])
            .stdin(Stdio::piped())
            .stdout(Stdio::from(std::fs::File::create(&destination)?))
            .spawn()?;
        child
            .stdin
            .take()
            .ok_or("gzip input absent")?
            .write_all(bytes)?;
        if !child.wait()?.success() {
            return Err("gzip failed".into());
        }
        destination
    };
    let entry = json!({"stored":destination.file_name().unwrap().to_string_lossy(),"original":path.file_name().unwrap().to_string_lossy(),"bytes":bytes.len(),"sha256":ariadne::reports::sha256(bytes)});
    writeln!(
        std::fs::OpenOptions::new()
            .create(true)
            .append(true)
            .open(path.parent().unwrap().join("artifact-identities.ndjson"))?,
        "{}",
        entry
    )?;
    Ok(destination)
}
fn freeze(path: &Path, expected: &str) -> Result<Value, Box<dyn Error>> {
    let bytes = std::fs::read(path)?;
    if ariadne::reports::sha256(&bytes) != expected {
        return Err(format!("reference identity differs: {}", path.display()).into());
    }
    Ok(serde_json::from_slice(&bytes)?)
}
fn entry_audit(
    path: &ariadne::bap::target_resolution::PathReceipt,
    entry: &ariadne::bap::target_resolution::Entry,
    table: &Value,
) -> Value {
    let expected = table["entries"].as_array().and_then(|rows| {
        rows.iter()
            .find(|e| e["index"].as_u64() == Some(entry.index))
    });
    let word = if entry.bytes_hex.len() == 8 {
        u32::from_str_radix(&entry.bytes_hex[6..8], 16)
            .ok()
            .zip(u32::from_str_radix(&entry.bytes_hex[4..6], 16).ok())
            .zip(u32::from_str_radix(&entry.bytes_hex[2..4], 16).ok())
            .zip(u32::from_str_radix(&entry.bytes_hex[0..2], 16).ok())
            .map(|(((a, b), c), d)| (a << 24) | (b << 16) | (c << 8) | d)
    } else {
        None
    };
    let base = address(&path.base).ok();
    let offset = word.map(|n| i64::from(n as i32));
    let numeric = base.zip(offset).map(|(b, o)| va(b.wrapping_add(o as u64)));
    let slot = base.map(|b| va(b.wrapping_add(entry.index.wrapping_mul(4))));
    json!({"path_root":path.root,"frontier":path.frontier,"form":path.form,"base":path.base,"index":entry.index,"slot":entry.slot,"bytes":entry.bytes_hex,"target":entry.target,"independent_signed_offset":offset,"independent_wrapped_target":numeric,"independent_expected_slot":slot,"numeric_target_matches":numeric.as_ref()==Some(&entry.target),"slot_matches":slot.as_ref()==Some(&entry.slot),"matches_frozen_table_base":table["base"].as_str()==Some(path.base.as_str()),"matches_frozen_index_target":expected.is_some_and(|e|e["target"].as_str()==Some(entry.target.as_str())),"matches_frozen_signed_offset":expected.and_then(|e|e["offset_i32"].as_i64())==offset})
}
fn state(view: &dyn AnalysisView) -> Value {
    let s = view.state();
    json!({"phase":format!("{:?}",s.phase),"visited":s.visited.len(),"decoded":s.decoded.len(),"edges":s.edges.len(),"obligations":s.obligations.iter().map(|o|json!({"site":va(o.site),"reason":format!("{:?}",o.reason)})).collect::<Vec<_>>(),"reaching_definitions":s.reaching.values().map(|d|d.len()).sum::<usize>(),"slice":s.slice.len()})
}

fn coverage(
    report: &ariadne::bap::target_resolution::Report,
    row: &ariadne::bap::target_resolution::Resolution,
) -> Result<Value, Box<dyn Error>> {
    let value = serde_json::to_value(&row.coverage)?;
    if value["kind"] != "inductive_dispatch" {
        return Ok(json!({"kind":value["kind"],"evidence":value}));
    }
    let proof = &value["receipt"];
    let count = |key: &str| proof[key].as_array().map(Vec::len);
    Ok(
        json!({"kind":"inductive_dispatch","schema":proof["schema"],"rule":proof["rule"],
        "snapshot_bound":proof["snapshot"]==report.snapshot,
        "artifact_bound":proof["artifact_sha256"]==report.artifact_sha256,
        "graph_bound":proof["graph_sha256"]==report.graph_sha256,
        "dispatch_bound":proof["dispatch"]==row.site,
        "model_sha256":proof["model_sha256"],"engine_sha256":proof["engine_sha256"],
        "region_sites":count("region"),"internal_edges":count("internal_edges"),
        "entrances":count("entrances"),"exits":count("exits"),
        "assertions":count("assertions"),"corridors":count("corridors"),
        "source_refs":count("source_refs"),"table":proof["table"],
        "scope":"Whole queried-dispatch coverage is carried by this receipt; raw receipt remains in target-report.json."}),
    )
}

fn reconcile(
    p: &FilePreparedAnalysis,
    witness: &Value,
    baseline: &Value,
) -> Result<Value, Box<dyn Error>> {
    let target = p.target_resolution().ok_or("target report absent")?;
    let composition = p.composition().ok_or("composition report absent")?;
    let candidate = ariadne::bap::composition::candidate_request(composition)?;
    let historical: BTreeSet<u64> = baseline["input"]["sites"]
        .as_array()
        .ok_or("baseline sites absent")?
        .iter()
        .map(|s| address(s["site"].as_str().ok_or("baseline site")?))
        .collect::<Result<_, _>>()?;
    let mut all_expected = BTreeSet::new();
    let mut all_actual = BTreeSet::new();
    let mut sites = Vec::new();
    let mut entries_total = 0usize;
    for table in witness["tables"]
        .as_array()
        .ok_or("witness tables absent")?
    {
        let site = table["jump"].as_str().ok_or("witness jump absent")?;
        let expected: BTreeSet<u64> = table["entries"]
            .as_array()
            .ok_or("witness entries absent")?
            .iter()
            .map(|e| address(e["target"].as_str().ok_or("witness target")?))
            .collect::<Result<_, _>>()?;
        entries_total += table["entries"].as_array().unwrap().len();
        all_expected.extend(&expected);
        let row = target.resolutions.iter().find(|r| r.site == site);
        let actual: BTreeSet<u64> = row
            .map(|r| {
                r.targets
                    .iter()
                    .map(|s| address(s))
                    .collect::<Result<_, _>>()
            })
            .transpose()?
            .unwrap_or_default();
        all_actual.extend(&actual);
        let entry_rows: Vec<_> = row
            .into_iter()
            .flat_map(|r| &r.paths)
            .flat_map(|path| {
                path.entries
                    .iter()
                    .map(move |e| entry_audit(path, e, table))
            })
            .collect();
        let missing: Vec<_> = expected.difference(&actual).map(|a| va(*a)).collect();
        let extra: Vec<_> = actual.difference(&expected).map(|a| va(*a)).collect();
        sites.push(json!({"site":site,"table_base":table["base"],"expected_entries":table["entries"].as_array().unwrap().len(),"expected_targets":expected.iter().map(|a|va(*a)).collect::<Vec<_>>(),"actual_targets":actual.iter().map(|a|va(*a)).collect::<Vec<_>>(),"receipt_present":row.is_some(),"complete":row.map(|r|r.complete),"unresolved":row.map(|r|&r.unresolved),"coverage":row.map(|r|coverage(target,r)).transpose()?,"path_count":row.map(|r|r.paths.len()),"path_summaries":row.map(|r|r.paths.iter().map(|p|json!({"root":p.root,"frontier":p.frontier,"form":p.form,"base":p.base,"indices":p.indices,"entry_count":p.entries.len(),"reverse_path":p.reverse_path,"reverse_edges":p.reverse_edges})).collect::<Vec<_>>()),"entry_witnesses":entry_rows,"missing_from_numeric_witness":missing,"additional_candidates":extra,"target_sets_equal":expected==actual,"all_expected_retained":expected.is_subset(&actual),"scope_note":"Missing numeric-witness candidates receive no index-exclusion credit in this harness; read/path receipts remain available for independent review."}));
    }
    let new: BTreeSet<_> = all_expected.difference(&historical).copied().collect();
    if sites.len() != 9 || entries_total != 370 || all_expected.len() != 64 || new.len() != 51 {
        return Err("frozen nine-table/baseline accounting differs".into());
    }
    let discovery: Vec<_> = all_expected.iter().map(|a|{
        let evidence = p.prepared.instructions.get(a);
        let control = candidate.instructions.get(a);
        json!({"site":va(*a),"historically_represented":historical.contains(a),"previously_absent":new.contains(a),"attempted":p.materialization.attempted.contains(a),"candidate_endpoint":candidate.addresses.contains(a),"candidate_decodable":candidate.decodable.contains(a),"final_decodable":p.prepared.request.decodable.contains(a),"control":control.map(|i|format!("{:?}",i.kind)),"opcode":evidence.and_then(|e|e.opcode.as_ref()),"quality":evidence.map(|e|format!("{:?}",e.quality)),"semantic_status":evidence.and_then(|e|e.semantic.as_ref()).map(|s|&s.status),"semantic_gaps":evidence.and_then(|e|e.semantic.as_ref()).map(|s|&s.gaps),"gaps":p.prepared.gaps.iter().filter(|g|g.address==*a).map(|g|format!("{:?}",g.reason)).collect::<Vec<_>>(),"unattempted_reason":if p.materialization.attempted.contains(a){None}else{Some("not-attempted; no exclusion credited")}})
    }).collect();
    Ok(
        json!({"schema":"ariadne.loop-aware-dump-table-reconciliation/v1","dump_sha256":DUMP,"witness_sha256":WITNESS,"baseline_sha256":BASELINE,"memory_premise":target.memory_premise,"all_original_target_sets_equal":sites.iter().all(|s|s["target_sets_equal"]==true),"complete_original_sites":sites.iter().filter(|s|s["complete"]==true).count(),"original_sites":sites,"expected_tables":9,"expected_entries":370,"expected_distinct_targets":64,"expected_previously_absent_starts":51,"actual_distinct_targets_at_original_sites":all_actual.len(),"missing_union_candidates":all_expected.difference(&all_actual).map(|a|va(*a)).collect::<Vec<_>>(),"additional_union_candidates":all_actual.difference(&all_expected).map(|a|va(*a)).collect::<Vec<_>>(),"previously_absent_attempted":new.intersection(&p.materialization.attempted).count(),"previously_absent_unattempted":new.difference(&p.materialization.attempted).map(|a|va(*a)).collect::<Vec<_>>(),"destination_discovery":discovery,"newly_exposed_indirects":target.resolutions.iter().filter(|r|!witness["tables"].as_array().unwrap().iter().any(|t|t["jump"]==r.site)).map(|r|json!({"site":r.site,"targets":r.targets,"complete":r.complete,"unresolved":r.unresolved,"coverage":serde_json::to_value(&r.coverage).ok()})).collect::<Vec<_>>() }),
    )
}

fn caller_audit(p: &FilePreparedAnalysis, root: &Path) -> Result<Value, Box<dyn Error>> {
    let fixture_path = root.join("tests/bap/fixtures/caller-coverage/dump-four-shapes.json");
    let old = freeze(&fixture_path, CALLER_EXPECTATION)?;
    let guard = p.guard_admission().ok_or("caller guard report absent")?;
    let flow = guard
        .input
        .normal_flow
        .as_ref()
        .ok_or("caller normal-flow scope absent")?;
    let load_site = "0x00007ff7744035c9";
    let original: BTreeSet<String> = old["guard_input"]["barriers"]
        .as_array()
        .ok_or("caller baseline barriers absent")?
        .iter()
        .map(|s| {
            s.as_str()
                .map(str::to_owned)
                .ok_or("caller barrier is not an address")
        })
        .collect::<Result<_, _>>()?;
    if original.len() != 30 || !original.contains(load_site) {
        return Err("caller baseline does not have the expected 30 barriers".into());
    }
    let remaining: BTreeSet<String> = guard.input.barriers.iter().cloned().collect();
    let terminals: BTreeSet<String> = flow.terminals.iter().map(|r| r.site.clone()).collect();
    let original_terminals: BTreeSet<String> = original
        .iter()
        .filter(|s| s.as_str() != load_site)
        .cloned()
        .collect();
    let receipts = flow.terminals.iter().filter(|r|original_terminals.contains(&r.site))
        .map(|r|json!({"site":r.site,"rule":r.rule,"bytes_hex":r.bytes_hex,"exception":r.exception,"annotation":r.annotation,"resumption":r.resumption}))
        .collect::<Vec<_>>();
    let int3_count = receipts
        .iter()
        .filter(|r| r["rule"] == "int3-exception/v1")
        .count();
    let roots_equal = serde_json::to_value(&guard.input.roots)? == old["guard_input"]["roots"];
    let premises_equal =
        serde_json::to_value(&guard.input.premises)? == old["guard_input"]["premises"];
    let load = guard.guards.iter().find(|g| g.site == load_site);
    Ok(json!({"schema":"ariadne.caller-normal-flow-dump-audit/v1",
        "snapshot":guard.input.snapshot,"artifact_sha256":guard.input.artifact_sha256,
        "expectation_source":fixture_path,"expectation_sha256":CALLER_EXPECTATION,
        "expectations_are_configuration":false,"scope":flow.scope,"normal_flow_kind":flow.kind,
        "original_barrier_count":original.len(),"original_barriers":original,
        "remaining_barriers":remaining,"all_original_barriers_removed":original.is_disjoint(&remaining),
        "no_remaining_barriers":remaining.is_empty(),"guard_stop":guard.stop,
        "all_original_terminal_receipts_checked":original_terminals.is_subset(&terminals)&&int3_count==27&&receipts.len()==29,
        "original_int3_receipts":int3_count,"original_terminal_receipts":receipts,
        "total_terminal_receipts":flow.terminals.len(),"root_evidence_equal":roots_equal,
        "premise_evidence_equal":premises_equal,"roots_and_premises_unchanged":roots_equal&&premises_equal,
        "load_guard":load.map(|g|json!({"site":g.site,"verdict":g.verdict,"residual":g.residual,"receipt_sha256":g.receipt_sha256})),
        "all_guard_outcomes":guard.guards.iter().map(|g|json!({"site":g.site,"verdict":g.verdict,"residual":g.residual})).collect::<Vec<_>>(),
        "claim_boundary":"Checked local normal completion; exception handler/debugger resumption and exceptional effects remain outside scope. Guard coverage, alignment verdict, completed-core parity and frame paths are separate outcomes."}))
}

fn main() -> Result<(), Box<dyn Error>> {
    let args: Vec<_> = std::env::args().skip(1).collect();
    if args.len() < 4 {
        return Err("usage: dump-probe DUMP OUT TABLE-WITNESS HISTORICAL-GUARD [paths]".into());
    }
    let out = PathBuf::from(&args[1]);
    std::fs::create_dir_all(&out)?;
    let started = Instant::now();
    let witness = freeze(Path::new(&args[2]), WITNESS)?;
    let baseline = freeze(Path::new(&args[3]), BASELINE)?;
    let snapshot = FileSnapshot::open_minidump(
        &args[0],
        OpenLimits {
            max_file_bytes: 2 * 1024 * 1024 * 1024,
            max_read_bytes: 32 * 1024 * 1024,
            ..Default::default()
        },
    )?;
    let md = snapshot.metadata();
    if md.artifact_sha256 != DUMP {
        return Err("dump identity differs".into());
    }
    let thread = md.exception.as_ref().ok_or("exception absent")?.thread_id;
    let chain = snapshot.recover_thread_frames(
        thread,
        FrameRecoveryLimits {
            max_frames: 6,
            max_runtime_entries: 440520,
            max_read_bytes: 6334816,
            ..Default::default()
        },
    )?;
    snapshot.validate_thread_frames(&chain)?;
    write(&out, "frames.json", &serde_json::to_value(&chain)?)?;
    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    let config = ariadne::bap::Config::from_env();
    let options = ariadne::effects::PreparationOptions::default();
    let decoder = root.join("target/caller-llvm/ariadne-llvm-mc");
    let limits = PrepareLimits {
        max_starts: 16384,
        max_candidates: 32768,
        max_prefix_bytes: 16384 * 15,
        max_decoder_batches: 16384,
        batch_size: 128,
    };
    let guard = ariadne::bap::guard_admission::Limits {
        max_sites: 32768,
        max_edges: 131072,
        max_residue_updates: 4194304,
        max_transfer_evaluations: 67108864,
        max_closure_rounds: 32768,
        max_evidence_bytes: 64 * 1024 * 1024,
        max_elapsed_ms: 120000,
    };
    let target = ariadne::bap::target_resolution::Limits {
        max_entries: 16384,
        max_targets: 65536,
        max_read_bytes: 16 * 1024 * 1024,
        max_evidence_bytes: 128 * 1024 * 1024,
        max_values: 4096,
        max_steps: 2097152,
        max_path_steps: 512,
        max_rounds: 64,
        max_elapsed_ms: 120000,
    };
    let mut native = CoreConfig::from_directory(root.join("target/caller-native"));
    native.timeout = Duration::from_secs(CORE_SECONDS);
    native.max_request_bytes = 128 * 1024 * 1024;
    let binding_path = out.join("source-binding.json");
    let binding: Value = serde_json::from_slice(&std::fs::read(&binding_path)?)?;
    if binding["composition_model_sha256"] != ariadne::bap::composition::model_identity()
        || binding["composition_engine_sha256"] != ariadne::bap::composition::checker_identity()
    {
        return Err(
            "linked release library differs from the recorded model/checker sources".into(),
        );
    }
    let mut summary = json!({"schema":"ariadne.caller-coverage-dump-probe/v1","dump_sha256":DUMP,"thread":thread,"frames_recovered":chain.frames.len(),"unwind_stop":chain.stop_reason,"source_binding_file":"source-binding.json","source_binding_sha256":ariadne::reports::sha256(&std::fs::read(&binding_path)?),"source_binding":binding,"preparation_limits":limits,"guard_limits":guard,"target_limits":target,"native_request_bytes":native.max_request_bytes,"native_timeout_seconds":CORE_SECONDS,"native_action_limit":1_000_000,"core_action_limit":CORE_ACTION_LIMIT,"rust_core_deadline_ms":CORE_SECONDS*1000,"native_finish_precondition":"Always attempted after native initialization, independently of Rust completion or decoded-node count; helper has a 1000000-action budget, transport has a finite 600s response timeout and runner has a finite 2400s total process timeout","binding_hard_limits":{"decoded_nodes":8192,"edges":32768},"premises":["FrozenCapturedControlData","windows-amd64-normal-call-entry/v1"],"scope":"Capture-only validated frame activations; table/loop fixtures are test expectations, never resolver configuration or extra roots; partial/unavailable outcomes remain explicit.","frames":[]});
    write(&out, "probe-summary.json", &summary)?;
    let setup = FramePreparation {
        decoder: &decoder,
        config: &config,
        options: &options,
        limits,
    };
    let mut bound = Vec::new();
    let indices = if args.get(4).is_some_and(|s| s == "paths") {
        vec![1usize, 0usize]
    } else {
        vec![1usize]
    };
    for index in indices {
        let frame_out = out.join(format!("frame-{index}"));
        std::fs::create_dir_all(&frame_out)?;
        let Some(frame) = chain.frames.get(index) else {
            summary["frames"]
                .as_array_mut()
                .unwrap()
                .push(json!({"frame":index,"status":"unavailable-frame"}));
            continue;
        };
        let Some(entry) = frame.function_begin else {
            summary["frames"]
                .as_array_mut()
                .unwrap()
                .push(json!({"frame":index,"status":"missing-function-entry"}));
            continue;
        };
        eprintln!("preparing frame {index} entry {}", va(entry));
        let prepared = match snapshot
            .prepare_frame_with_bap_assuming_windows_call_entry_and_target_resolution(
                &chain,
                index,
                entry,
                &setup,
                guard.clone(),
                target.clone(),
            ) {
            Ok(p) => p,
            Err(e) => {
                let mut failure = json!({"frame":index,"entry":va(entry),"status":"preparation-failed","error":e.to_string()});
                if let Some(PreparationError::LimitReached { reason, progress }) =
                    e.downcast_ref::<PreparationError>()
                {
                    failure["limit"] = json!({"reason":reason,"attempted":progress.attempted.len(),"prefix_bytes":progress.prefix_bytes,"batches":progress.decoder_batches});
                }
                summary["frames"].as_array_mut().unwrap().push(failure);
                write(&out, "probe-summary.json", &summary)?;
                continue;
            }
        };
        let g = prepared.guard_admission().ok_or("paired guard missing")?;
        let t = prepared
            .target_resolution()
            .ok_or("paired target missing")?;
        let c = prepared.composition().ok_or("composition missing")?;
        write(&frame_out, "guard-report.json", &serde_json::to_value(g)?)?;
        write(&frame_out, "target-report.json", &serde_json::to_value(t)?)?;
        write(
            &frame_out,
            "composition-report.json",
            &serde_json::to_value(c)?,
        )?;
        write(&frame_out, "prepared-full.json", &preparation(&prepared)?)?;
        if index == 1 {
            let reconciliation = reconcile(&prepared, &witness, &baseline)?;
            write(&out, "nine-table-reconciliation.json", &reconciliation)?;
            summary["original_table_gates"] = json!({"all_nine_target_sets_equal":reconciliation["all_original_target_sets_equal"],"original_complete_sites":reconciliation["complete_original_sites"],"previously_missing_attempted":reconciliation["previously_absent_attempted"],"previously_missing_unattempted":reconciliation["previously_absent_unattempted"]});
            let audit = loop_audit(&prepared, &root)?;
            summary["loop_closure"] = audit;
            let caller = caller_audit(&prepared, &root)?;
            write(&out, "caller-normal-flow-audit.json", &caller)?;
            summary["caller_normal_flow"] = caller;
        }
        let mut row = json!({"frame":index,"entry":va(entry),"status":"prepared","composition_schema":c.schema,"composition_profile":c.profile,"target_schema":t.schema,"target_profile":t.profile,"candidate_sites":g.input.sites.len(),"attempted_starts":prepared.materialization.attempted.len(),"candidate_epoch":c.candidate_epoch,"guard_stop":g.stop,"target_stop":t.stop,"guard_barriers":g.input.barriers,"guards":g.guards.iter().map(|x|json!({"site":x.site,"verdict":x.verdict,"residual":x.residual})).collect::<Vec<_>>(),"target_sites":t.resolutions.len(),"complete_target_sites":t.resolutions.iter().filter(|r|r.complete).count(),"guard_report_bytes":serde_json::to_vec(g)?.len(),"target_report_bytes":serde_json::to_vec(t)?.len(),"composition_report_bytes":serde_json::to_vec(c)?.len()});
        let core_begin = Instant::now();
        let mut rust = Analyzer::new(prepared.prepared.request.clone())?;
        let mut actions = 0usize;
        while rust.state().phase != Phase::Done
            && actions < CORE_ACTION_LIMIT
            && core_begin.elapsed() < Duration::from_secs(CORE_SECONDS)
        {
            if !rust.step() {
                break;
            }
            actions += 1;
        }
        row["rust_core"] = json!({"actions":actions,"elapsed_ms":core_begin.elapsed().as_millis(),"observation":state(&rust),"completed":rust.state().phase==Phase::Done});
        let completed = if rust.state().phase == Phase::Done {
            Some(CompletedAnalysis::from_analyzer(rust))
        } else {
            None
        };
        let native_begin = Instant::now();
        match NativeAnalyzer::from_capture(
            &native,
            &format!("caller-coverage-dump-frame-{index}"),
            &prepared,
        ) {
            Ok(n) => {
                row["native_initialize"] = json!({"status":"accepted","elapsed_ms":native_begin.elapsed().as_millis(),"identity":n.identity()});
                match n.complete() {
                    Ok(done) => {
                        let reference = completed.as_ref();
                        row["native_core"] = json!({"status":"completed","elapsed_ms":native_begin.elapsed().as_millis(),"observation":state(&done),"rust_reference_completed":reference.is_some(),"request_equal":reference.map(|r|r.request()==done.request()),"full_state_equal":reference.map(|r|r.state()==done.state())});
                        save(
                            &frame_out.join("minidump-native.json"),
                            ariadne::input::report::render_bound(
                                &prepared,
                                &done,
                                ariadne::input::report::ReportFormat::Json,
                                false,
                            )?
                            .as_bytes(),
                        )?;
                    }
                    Err(e) => {
                        row["native_core"] = json!({"status":"failed","elapsed_ms":native_begin.elapsed().as_millis(),"error":e.to_string()});
                    }
                }
            }
            Err(e) => {
                row["native_initialize"] = json!({"status":"rejected","elapsed_ms":native_begin.elapsed().as_millis(),"error":e.to_string()})
            }
        }
        if let Some(done) = completed {
            save(
                &frame_out.join("minidump-rust.json"),
                ariadne::input::report::render_bound(
                    &prepared,
                    &done,
                    ariadne::input::report::ReportFormat::Json,
                    false,
                )?
                .as_bytes(),
            )?;
            match ariadne::input::investigation::bind_investigation(&prepared, &done) {
                Ok(b) => bound.push((index, b)),
                Err(e) => {
                    row["investigation_binding"] = json!({"status":"failed","error":e.to_string()})
                }
            }
        }
        write(&frame_out, "outcome.json", &row)?;
        if index == 1 {
            let accepted = row["native_initialize"]["status"] == "accepted";
            summary["loop_closure"]["paired_native_admission"] = json!({"accepted":accepted,"native_initialize":row["native_initialize"],"request_equal":row["native_core"]["request_equal"],"full_state_equal":row["native_core"]["full_state_equal"],"scope":"Native paired admission independently checks every included loop receipt; completed full-state equality is recorded separately."});
            write(&out, "loop-closure-audit.json", &summary["loop_closure"])?;
        }
        summary["frames"].as_array_mut().unwrap().push(row);
        write(&out, "probe-summary.json", &summary)?;
    }
    if args.get(4).is_some_and(|s| s == "paths") {
        let path_limits = FramePathLimits {
            max_graph_nodes: 8192,
            max_graph_edges: 32768,
            max_search_states: 65536,
            max_scanned_edges: 262144,
            max_path_steps: 4096,
            max_node_visits: 2,
            max_frontier_bytes: 32 * 1024 * 1024,
            max_output_bytes: 128 * 1024 * 1024,
            deadline_ms: 10000,
        };
        summary["path_limits"] = json!(path_limits);
        match ariadne::investigation::bind_frame_paths(chain, bound).and_then(|b| {
            ariadne::investigation::list_frame_paths(
                &b,
                FramePathQuestion {
                    from_frame: 1,
                    to_frame: 0,
                    np: 10,
                },
                path_limits,
            )
        }) {
            Ok(listing) => {
                let listing_json: Value =
                    serde_json::from_str(&ariadne::reports::render_frame_paths(
                        &listing,
                        ariadne::reports::Format::Json,
                    )?)?;
                for (name, format) in [
                    ("frame-paths.json", ariadne::reports::Format::Json),
                    ("frame-paths.txt", ariadne::reports::Format::Text),
                    ("frame-paths.dot", ariadne::reports::Format::Dot),
                ] {
                    save(
                        &out.join(name),
                        ariadne::reports::render_frame_paths(&listing, format)?.as_bytes(),
                    )?;
                }
                summary["frame_paths"] = json!({"status":"listed","json":"frame-paths.json or lossless frame-paths.json.gz; see artifact-identities.ndjson","available":listing_json["available"],"coverage":listing_json["coverage"],"returned_count":listing_json["returned_count"],"stop_reason":listing_json["stop_reason"],"truncated":listing_json["truncated"],"enumeration_exhausted":listing_json["enumeration_exhausted"],"counters":listing_json["counters"]});
            }
            Err(e) => {
                summary["frame_paths"] = json!({"status":"unavailable","error":e.to_string()})
            }
        }
    } else {
        summary["frame_paths"] =
            json!({"status":"unexercised","reason":"paths mode not requested"});
    }
    summary["elapsed_ms"] = json!(started.elapsed().as_millis());
    let loop_sites = summary["loop_closure"]["sites"].as_array();
    let loops_proved = loop_sites.is_some_and(|sites| {
        sites.len() == 2
            && sites.iter().all(|site| {
                site["complete"] == true
                    && site["target_sets_equal"] == true
                    && site["coverage"]["kind"] == "inductive_dispatch"
                    && site["coverage"]["snapshot_bound"] == true
                    && site["coverage"]["artifact_bound"] == true
                    && site["coverage"]["graph_bound"] == true
                    && site["coverage"]["dispatch_bound"] == true
            })
    });
    summary["acceptance_observations"] = json!({
        "all_original_table_sets_equal":summary["original_table_gates"]["all_nine_target_sets_equal"],
        "all_nine_original_sites_complete":summary["original_table_gates"]["original_complete_sites"]==9,
        "all_51_formerly_missing_starts_attempted":summary["original_table_gates"]["previously_missing_attempted"]==51,
        "both_targeted_loop_receipts_complete_and_bound":loops_proved,
        "paired_native_accepted_loop_receipts":loops_proved && summary["loop_closure"]["paired_native_admission"]["accepted"]==true,
        "all_30_original_barriers_removed":summary["caller_normal_flow"]["all_original_barriers_removed"],
        "no_remaining_caller_barriers":summary["caller_normal_flow"]["no_remaining_barriers"],
        "all_29_original_terminal_receipts_checked":summary["caller_normal_flow"]["all_original_terminal_receipts_checked"],
        "caller_roots_and_premises_unchanged":summary["caller_normal_flow"]["roots_and_premises_unchanged"],
        "frames_with_full_native_rust_parity":summary["frames"].as_array().unwrap().iter()
          .filter(|frame|frame["native_core"]["request_equal"]==true && frame["native_core"]["full_state_equal"]==true)
          .map(|frame|frame["frame"].clone()).collect::<Vec<_>>(),
        "frame_paths_available":summary["frame_paths"]["available"],
        "frame_path_coverage":summary["frame_paths"]["coverage"],
        "frame_path_stop_reason":summary["frame_paths"]["stop_reason"],
        "scope":"Loop closure, candidate recovery, guard coverage, completed-core parity and path availability are separate claims; unavailable paths never establish no possible path."});
    write(&out, "probe-summary.json", &summary)?;
    println!("{}", serde_json::to_string_pretty(&summary)?);
    Ok(())
}
