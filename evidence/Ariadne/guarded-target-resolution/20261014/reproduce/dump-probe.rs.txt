const DUMP: &str = "9be21e47453fec954857db52db7dfe3b8c5a5c6a49369d39a10eb701de9624ab";
const WITNESS: &str = "1390abca21bb7e2db9cde9a07067f60f5493498a7866fb648dcb103994674baf";
const BASELINE: &str = "d0294c9b940d8366cb89d78a95d2186c89196dbf1a7955e70189c22afa2abda0";

fn va(a: u64) -> String {
    format!("0x{a:016x}")
}
fn address(s: &str) -> Result<u64, Box<dyn Error>> {
    Ok(u64::from_str_radix(s.trim_start_matches("0x"), 16)?)
}
fn write(out: &Path, name: &str, value: &Value) -> Result<(), Box<dyn Error>> {
    save(&out.join(name), &serde_json::to_vec_pretty(value)?)?;
    Ok(())
}
fn save(path: &Path, bytes: &[u8]) -> Result<PathBuf, Box<dyn Error>> {
    let destination = if bytes.len() < 1024 * 1024 {
        std::fs::write(path, bytes)?;
        path.to_path_buf()
    } else {
        let mut name = path.as_os_str().to_os_string();
        name.push(".gz");
        let destination = PathBuf::from(name);
        let mut child = Command::new("gzip")
            .args(["-9", "-c"])
            .stdin(Stdio::piped())
            .stdout(Stdio::from(std::fs::File::create(&destination)?))
            .spawn()?;
        child
            .stdin
            .take()
            .ok_or("gzip input absent")?
            .write_all(bytes)?;
        if !child.wait()?.success() {
            return Err("gzip failed".into());
        }
        destination
    };
    let entry = json!({"stored":destination.file_name().unwrap().to_string_lossy(),"original":path.file_name().unwrap().to_string_lossy(),"bytes":bytes.len(),"sha256":ariadne::reports::sha256(bytes)});
    writeln!(
        std::fs::OpenOptions::new()
            .create(true)
            .append(true)
            .open(path.parent().unwrap().join("artifact-identities.ndjson"))?,
        "{}",
        entry
    )?;
    Ok(destination)
}
fn freeze(path: &Path, expected: &str) -> Result<Value, Box<dyn Error>> {
    let bytes = std::fs::read(path)?;
    if ariadne::reports::sha256(&bytes) != expected {
        return Err(format!("reference identity differs: {}", path.display()).into());
    }
    Ok(serde_json::from_slice(&bytes)?)
}
fn entry_audit(
    path: &ariadne::bap::target_resolution::PathReceipt,
    entry: &ariadne::bap::target_resolution::Entry,
    table: &Value,
) -> Value {
    let expected = table["entries"].as_array().and_then(|rows| {
        rows.iter()
            .find(|e| e["index"].as_u64() == Some(entry.index))
    });
    let word = if entry.bytes_hex.len() == 8 {
        u32::from_str_radix(&entry.bytes_hex[6..8], 16)
            .ok()
            .zip(u32::from_str_radix(&entry.bytes_hex[4..6], 16).ok())
            .zip(u32::from_str_radix(&entry.bytes_hex[2..4], 16).ok())
            .zip(u32::from_str_radix(&entry.bytes_hex[0..2], 16).ok())
            .map(|(((a, b), c), d)| (a << 24) | (b << 16) | (c << 8) | d)
    } else {
        None
    };
    let base = address(&path.base).ok();
    let offset = word.map(|n| i64::from(n as i32));
    let numeric = base.zip(offset).map(|(b, o)| va(b.wrapping_add(o as u64)));
    let slot = base.map(|b| va(b.wrapping_add(entry.index.wrapping_mul(4))));
    json!({"path_root":path.root,"frontier":path.frontier,"form":path.form,"base":path.base,"index":entry.index,"slot":entry.slot,"bytes":entry.bytes_hex,"target":entry.target,"independent_signed_offset":offset,"independent_wrapped_target":numeric,"independent_expected_slot":slot,"numeric_target_matches":numeric.as_ref()==Some(&entry.target),"slot_matches":slot.as_ref()==Some(&entry.slot),"matches_frozen_table_base":table["base"].as_str()==Some(path.base.as_str()),"matches_frozen_index_target":expected.is_some_and(|e|e["target"].as_str()==Some(entry.target.as_str())),"matches_frozen_signed_offset":expected.and_then(|e|e["offset_i32"].as_i64())==offset})
}
fn state(view: &dyn AnalysisView) -> Value {
    let s = view.state();
    json!({"phase":format!("{:?}",s.phase),"visited":s.visited.len(),"decoded":s.decoded.len(),"edges":s.edges.len(),"obligations":s.obligations.iter().map(|o|json!({"site":va(o.site),"reason":format!("{:?}",o.reason)})).collect::<Vec<_>>(),"reaching_definitions":s.reaching.values().map(|d|d.len()).sum::<usize>(),"slice":s.slice.len()})
}
fn preparation(p: &FilePreparedAnalysis) -> Result<Value, Box<dyn Error>> {
    let m = &p.materialization;
    Ok(
        json!({"schema":"ariadne.composition-dump-preparation-probe/v1","snapshot":p.snapshot.snapshot_id,"artifact_sha256":p.snapshot.artifact_sha256,"query_id":m.query_identity,"normalized":ariadne::bap::core_adapter::encode_request(&p.prepared.request)?,"materialization":{"roots":m.entry_points.iter().map(|a|va(*a)).collect::<Vec<_>>(),"seeds":m.slice_seeds.iter().map(|a|va(*a)).collect::<Vec<_>>(),"limits":m.limits,"attempted":m.attempted.iter().map(|a|va(*a)).collect::<Vec<_>>(),"referenced":m.referenced.iter().map(|a|va(*a)).collect::<Vec<_>>(),"unattempted":m.unattempted_references.iter().map(|a|va(*a)).collect::<Vec<_>>(),"prefix_bytes":m.prefix_bytes,"decoder_batches":m.decoder_batches},"sites":p.prepared.instructions.iter().map(|(a,e)|json!({"site":va(*a),"bytes":e.bytes.iter().map(|b|format!("{b:02x}")).collect::<String>(),"opcode":e.opcode,"quality":format!("{:?}",e.quality),"rule":e.rule,"semantic":e.semantic.as_ref().map(|s|json!({"backend":s.backend,"status":s.status,"projection":s.projection,"ast_sha256":s.ast_sha256,"gaps":s.gaps,"helper_sha256":s.helper_sha256,"runtime_sha256":s.runtime_sha256}))})).collect::<Vec<_>>(),"gaps":p.prepared.gaps.iter().map(|g|json!({"site":va(g.address),"reason":format!("{:?}",g.reason)})).collect::<Vec<_>>() }),
    )
}
fn reconcile(
    p: &FilePreparedAnalysis,
    witness: &Value,
    baseline: &Value,
) -> Result<Value, Box<dyn Error>> {
    let target = p.target_resolution().ok_or("target report absent")?;
    let composition = p.composition().ok_or("composition report absent")?;
    let candidate = ariadne::bap::composition::candidate_request(composition)?;
    let historical: BTreeSet<u64> = baseline["input"]["sites"]
        .as_array()
        .ok_or("baseline sites absent")?
        .iter()
        .map(|s| address(s["site"].as_str().ok_or("baseline site")?))
        .collect::<Result<_, _>>()?;
    let mut all_expected = BTreeSet::new();
    let mut all_actual = BTreeSet::new();
    let mut sites = Vec::new();
    let mut entries_total = 0usize;
    for table in witness["tables"]
        .as_array()
        .ok_or("witness tables absent")?
    {
        let site = table["jump"].as_str().ok_or("witness jump absent")?;
        let expected: BTreeSet<u64> = table["entries"]
            .as_array()
            .ok_or("witness entries absent")?
            .iter()
            .map(|e| address(e["target"].as_str().ok_or("witness target")?))
            .collect::<Result<_, _>>()?;
        entries_total += table["entries"].as_array().unwrap().len();
        all_expected.extend(&expected);
        let row = target.resolutions.iter().find(|r| r.site == site);
        let actual: BTreeSet<u64> = row
            .map(|r| {
                r.targets
                    .iter()
                    .map(|s| address(s))
                    .collect::<Result<_, _>>()
            })
            .transpose()?
            .unwrap_or_default();
        all_actual.extend(&actual);
        let entry_rows: Vec<_> = row
            .into_iter()
            .flat_map(|r| &r.paths)
            .flat_map(|path| {
                path.entries
                    .iter()
                    .map(move |e| entry_audit(path, e, table))
            })
            .collect();
        let missing: Vec<_> = expected.difference(&actual).map(|a| va(*a)).collect();
        let extra: Vec<_> = actual.difference(&expected).map(|a| va(*a)).collect();
        sites.push(json!({"site":site,"table_base":table["base"],"expected_entries":table["entries"].as_array().unwrap().len(),"expected_targets":expected.iter().map(|a|va(*a)).collect::<Vec<_>>(),"actual_targets":actual.iter().map(|a|va(*a)).collect::<Vec<_>>(),"receipt_present":row.is_some(),"complete":row.map(|r|r.complete),"unresolved":row.map(|r|&r.unresolved),"path_count":row.map(|r|r.paths.len()),"path_summaries":row.map(|r|r.paths.iter().map(|p|json!({"root":p.root,"frontier":p.frontier,"form":p.form,"base":p.base,"indices":p.indices,"entry_count":p.entries.len(),"reverse_path":p.reverse_path,"reverse_edges":p.reverse_edges})).collect::<Vec<_>>()),"entry_witnesses":entry_rows,"missing_from_numeric_witness":missing,"additional_candidates":extra,"all_expected_retained":expected.is_subset(&actual),"scope_note":"Missing numeric-witness candidates receive no index-exclusion credit in this harness; read/path receipts remain available for independent review."}));
    }
    let new: BTreeSet<_> = all_expected.difference(&historical).copied().collect();
    if sites.len() != 9 || entries_total != 370 || all_expected.len() != 64 || new.len() != 51 {
        return Err("frozen nine-table/baseline accounting differs".into());
    }
    let discovery: Vec<_> = all_expected.iter().map(|a|{
        let evidence = p.prepared.instructions.get(a);
        let control = candidate.instructions.get(a);
        json!({"site":va(*a),"historically_represented":historical.contains(a),"previously_absent":new.contains(a),"attempted":p.materialization.attempted.contains(a),"candidate_endpoint":candidate.addresses.contains(a),"candidate_decodable":candidate.decodable.contains(a),"final_decodable":p.prepared.request.decodable.contains(a),"control":control.map(|i|format!("{:?}",i.kind)),"opcode":evidence.and_then(|e|e.opcode.as_ref()),"quality":evidence.map(|e|format!("{:?}",e.quality)),"semantic_status":evidence.and_then(|e|e.semantic.as_ref()).map(|s|&s.status),"semantic_gaps":evidence.and_then(|e|e.semantic.as_ref()).map(|s|&s.gaps),"gaps":p.prepared.gaps.iter().filter(|g|g.address==*a).map(|g|format!("{:?}",g.reason)).collect::<Vec<_>>(),"unattempted_reason":if p.materialization.attempted.contains(a){None}else{Some("not-attempted; no exclusion credited")}})
    }).collect();
    Ok(
        json!({"schema":"ariadne.composition-dump-table-reconciliation/v1","dump_sha256":DUMP,"witness_sha256":WITNESS,"baseline_sha256":BASELINE,"memory_premise":target.memory_premise,"original_sites":sites,"expected_tables":9,"expected_entries":370,"expected_distinct_targets":64,"expected_previously_absent_starts":51,"actual_distinct_targets_at_original_sites":all_actual.len(),"missing_union_candidates":all_expected.difference(&all_actual).map(|a|va(*a)).collect::<Vec<_>>(),"additional_union_candidates":all_actual.difference(&all_expected).map(|a|va(*a)).collect::<Vec<_>>(),"previously_absent_attempted":new.intersection(&p.materialization.attempted).count(),"previously_absent_unattempted":new.difference(&p.materialization.attempted).map(|a|va(*a)).collect::<Vec<_>>(),"destination_discovery":discovery,"newly_exposed_indirects":target.resolutions.iter().filter(|r|!witness["tables"].as_array().unwrap().iter().any(|t|t["jump"]==r.site)).map(|r|json!({"site":r.site,"targets":r.targets,"complete":r.complete,"unresolved":r.unresolved})).collect::<Vec<_>>() }),
    )
}

fn main() -> Result<(), Box<dyn Error>> {
    let args: Vec<_> = std::env::args().skip(1).collect();
    if args.len() < 4 {
        return Err("usage: dump-probe DUMP OUT TABLE-WITNESS HISTORICAL-GUARD [paths]".into());
    }
    let out = PathBuf::from(&args[1]);
    std::fs::create_dir_all(&out)?;
    let started = Instant::now();
    let witness = freeze(Path::new(&args[2]), WITNESS)?;
    let baseline = freeze(Path::new(&args[3]), BASELINE)?;
    let snapshot = FileSnapshot::open_minidump(
        &args[0],
        OpenLimits {
            max_file_bytes: 2 * 1024 * 1024 * 1024,
            max_read_bytes: 32 * 1024 * 1024,
            ..Default::default()
        },
    )?;
    let md = snapshot.metadata();
    if md.artifact_sha256 != DUMP {
        return Err("dump identity differs".into());
    }
    let thread = md.exception.as_ref().ok_or("exception absent")?.thread_id;
    let chain = snapshot.recover_thread_frames(
        thread,
        FrameRecoveryLimits {
            max_frames: 6,
            max_runtime_entries: 440520,
            max_read_bytes: 6334816,
            ..Default::default()
        },
    )?;
    snapshot.validate_thread_frames(&chain)?;
    write(&out, "frames.json", &serde_json::to_value(&chain)?)?;
    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    let config = ariadne::bap::Config::from_env();
    let options = ariadne::effects::PreparationOptions::default();
    let decoder = root.join("target/ariadne-llvm-mc");
    let limits = PrepareLimits {
        max_starts: 16384,
        max_candidates: 32768,
        max_prefix_bytes: 16384 * 15,
        max_decoder_batches: 16384,
        batch_size: 128,
    };
    let guard = ariadne::bap::guard_admission::Limits {
        max_sites: 32768,
        max_edges: 131072,
        max_residue_updates: 4194304,
        max_transfer_evaluations: 67108864,
        max_closure_rounds: 32768,
        max_evidence_bytes: 64 * 1024 * 1024,
        max_elapsed_ms: 120000,
    };
    let target = ariadne::bap::target_resolution::Limits {
        max_entries: 16384,
        max_targets: 65536,
        max_read_bytes: 16 * 1024 * 1024,
        max_evidence_bytes: 128 * 1024 * 1024,
        max_values: 4096,
        max_steps: 2097152,
        max_path_steps: 512,
        max_rounds: 64,
        max_elapsed_ms: 120000,
    };
    let mut native = CoreConfig::from_directory(root.join("target/composed-native"));
    native.timeout = Duration::from_secs(120);
    native.max_request_bytes = 128 * 1024 * 1024;
    let mut summary = json!({"schema":"ariadne.composition-dump-probe/v1","dump_sha256":DUMP,"thread":thread,"frames_recovered":chain.frames.len(),"unwind_stop":chain.stop_reason,"preparation_limits":limits,"guard_limits":guard,"target_limits":target,"native_request_bytes":native.max_request_bytes,"native_timeout_seconds":120,"core_action_limit":100000,"rust_core_deadline_ms":30000,"native_finish_precondition":"Rust completed and final decoded nodes <=4096; helper retains its own finite action/response policies","binding_hard_limits":{"decoded_nodes":8192,"edges":32768},"premises":["FrozenCapturedControlData","windows-amd64-normal-call-entry/v1"],"scope":"Capture-only validated frame activations; table witness is test expectation, never resolver configuration or extra roots; partial/unavailable outcomes remain explicit.","frames":[]});
    write(&out, "probe-summary.json", &summary)?;
    let setup = FramePreparation {
        decoder: &decoder,
        config: &config,
        options: &options,
        limits,
    };
    let mut bound = Vec::new();
    let indices = if args.get(4).is_some_and(|s| s == "paths") {
        vec![1usize, 0usize]
    } else {
        vec![1usize]
    };
    for index in indices {
        let frame_out = out.join(format!("frame-{index}"));
        std::fs::create_dir_all(&frame_out)?;
        let Some(frame) = chain.frames.get(index) else {
            summary["frames"]
                .as_array_mut()
                .unwrap()
                .push(json!({"frame":index,"status":"unavailable-frame"}));
            continue;
        };
        let Some(entry) = frame.function_begin else {
            summary["frames"]
                .as_array_mut()
                .unwrap()
                .push(json!({"frame":index,"status":"missing-function-entry"}));
            continue;
        };
        eprintln!("preparing frame {index} entry {}", va(entry));
        let prepared = match snapshot
            .prepare_frame_with_bap_assuming_windows_call_entry_and_target_resolution(
                &chain,
                index,
                entry,
                &setup,
                guard.clone(),
                target.clone(),
            ) {
            Ok(p) => p,
            Err(e) => {
                let mut failure = json!({"frame":index,"entry":va(entry),"status":"preparation-failed","error":e.to_string()});
                if let Some(PreparationError::LimitReached { reason, progress }) =
                    e.downcast_ref::<PreparationError>()
                {
                    failure["limit"] = json!({"reason":reason,"attempted":progress.attempted.len(),"prefix_bytes":progress.prefix_bytes,"batches":progress.decoder_batches});
                }
                summary["frames"].as_array_mut().unwrap().push(failure);
                write(&out, "probe-summary.json", &summary)?;
                continue;
            }
        };
        let g = prepared.guard_admission().ok_or("paired guard missing")?;
        let t = prepared
            .target_resolution()
            .ok_or("paired target missing")?;
        let c = prepared.composition().ok_or("composition missing")?;
        write(&frame_out, "guard-report.json", &serde_json::to_value(g)?)?;
        write(&frame_out, "target-report.json", &serde_json::to_value(t)?)?;
        write(
            &frame_out,
            "composition-report.json",
            &serde_json::to_value(c)?,
        )?;
        write(&frame_out, "prepared-full.json", &preparation(&prepared)?)?;
        if index == 1 {
            write(
                &out,
                "nine-table-reconciliation.json",
                &reconcile(&prepared, &witness, &baseline)?,
            )?;
        }
        let mut row = json!({"frame":index,"entry":va(entry),"status":"prepared","candidate_sites":g.input.sites.len(),"attempted_starts":prepared.materialization.attempted.len(),"candidate_epoch":c.candidate_epoch,"guard_stop":g.stop,"target_stop":t.stop,"guard_barriers":g.input.barriers,"guards":g.guards.iter().map(|x|json!({"site":x.site,"verdict":x.verdict,"residual":x.residual})).collect::<Vec<_>>(),"target_sites":t.resolutions.len(),"complete_target_sites":t.resolutions.iter().filter(|r|r.complete).count(),"guard_report_bytes":serde_json::to_vec(g)?.len(),"target_report_bytes":serde_json::to_vec(t)?.len(),"composition_report_bytes":serde_json::to_vec(c)?.len()});
        let core_begin = Instant::now();
        let mut rust = Analyzer::new(prepared.prepared.request.clone())?;
        let mut actions = 0usize;
        while rust.state().phase != Phase::Done
            && actions < 100000
            && core_begin.elapsed() < Duration::from_secs(30)
        {
            if !rust.step() {
                break;
            }
            actions += 1;
        }
        row["rust_core"] = json!({"actions":actions,"elapsed_ms":core_begin.elapsed().as_millis(),"observation":state(&rust),"completed":rust.state().phase==Phase::Done});
        let completed = if rust.state().phase == Phase::Done {
            Some(CompletedAnalysis::from_analyzer(rust))
        } else {
            None
        };
        let native_begin = Instant::now();
        match NativeAnalyzer::from_capture(
            &native,
            &format!("combined-dump-frame-{index}"),
            &prepared,
        ) {
            Ok(n) => {
                row["native_initialize"] = json!({"status":"accepted","elapsed_ms":native_begin.elapsed().as_millis(),"identity":n.identity()});
                if let Some(reference) = completed
                    .as_ref()
                    .filter(|r| r.state().decoded.len() <= 4096)
                {
                    match n.complete() {
                        Ok(done) => {
                            row["native_core"] = json!({"status":"completed","observation":state(&done),"request_equal":reference.request()==done.request(),"full_state_equal":reference.state()==done.state()});
                            save(
                                &frame_out.join("minidump-native.json"),
                                ariadne::input::report::render_bound(
                                    &prepared,
                                    &done,
                                    ariadne::input::report::ReportFormat::Json,
                                    false,
                                )?
                                .as_bytes(),
                            )?;
                        }
                        Err(e) => {
                            row["native_core"] = json!({"status":"failed","error":e.to_string()})
                        }
                    }
                } else {
                    row["native_core"] = json!({"status":"unexercised","reason":"Rust did not finish within finite probe budget or decoded graph exceeded4096"});
                }
            }
            Err(e) => {
                row["native_initialize"] = json!({"status":"rejected","elapsed_ms":native_begin.elapsed().as_millis(),"error":e.to_string()})
            }
        }
        if let Some(done) = completed {
            save(
                &frame_out.join("minidump-rust.json"),
                ariadne::input::report::render_bound(
                    &prepared,
                    &done,
                    ariadne::input::report::ReportFormat::Json,
                    false,
                )?
                .as_bytes(),
            )?;
            match ariadne::input::investigation::bind_investigation(&prepared, &done) {
                Ok(b) => bound.push((index, b)),
                Err(e) => {
                    row["investigation_binding"] = json!({"status":"failed","error":e.to_string()})
                }
            }
        }
        write(&frame_out, "outcome.json", &row)?;
        summary["frames"].as_array_mut().unwrap().push(row);
        write(&out, "probe-summary.json", &summary)?;
    }
    if args.get(4).is_some_and(|s| s == "paths") {
        let path_limits = FramePathLimits {
            max_graph_nodes: 8192,
            max_graph_edges: 32768,
            max_search_states: 65536,
            max_scanned_edges: 262144,
            max_path_steps: 4096,
            max_node_visits: 2,
            max_frontier_bytes: 32 * 1024 * 1024,
            max_output_bytes: 128 * 1024 * 1024,
            deadline_ms: 10000,
        };
        summary["path_limits"] = json!(path_limits);
        match ariadne::investigation::bind_frame_paths(chain, bound).and_then(|b| {
            ariadne::investigation::list_frame_paths(
                &b,
                FramePathQuestion {
                    from_frame: 1,
                    to_frame: 0,
                    np: 10,
                },
                path_limits,
            )
        }) {
            Ok(listing) => {
                for (name, format) in [
                    ("frame-paths.json", ariadne::reports::Format::Json),
                    ("frame-paths.txt", ariadne::reports::Format::Text),
                    ("frame-paths.dot", ariadne::reports::Format::Dot),
                ] {
                    save(
                        &out.join(name),
                        ariadne::reports::render_frame_paths(&listing, format)?.as_bytes(),
                    )?;
                }
                summary["frame_paths"] = json!({"status":"listed","json":"frame-paths.json or lossless frame-paths.json.gz; see artifact-identities.ndjson"});
            }
            Err(e) => {
                summary["frame_paths"] = json!({"status":"unavailable","error":e.to_string()})
            }
        }
    } else {
        summary["frame_paths"] =
            json!({"status":"unexercised","reason":"paths mode not requested"});
    }
    summary["elapsed_ms"] = json!(started.elapsed().as_millis());
    write(&out, "probe-summary.json", &summary)?;
    println!("{}", serde_json::to_string_pretty(&summary)?);
    Ok(())
}
